// lib/today/playPlan.ts
/**
 * What the child plans for free time, per date — remembered in THIS browser only (parent
 * decision 08/11/2026: no server, parents never see it). Broken or missing storage = no plan.
 */
export const PLAY_PLAN_KEY = "play-plans";
const KEEP = 31;

function readAll(): Record<string, string> {
  try {
    const raw = (globalThis as { window?: { localStorage?: Storage } }).window?.localStorage?.getItem(PLAY_PLAN_KEY);
    const v = raw ? (JSON.parse(raw) as unknown) : {};
    if (!v || typeof v !== "object" || Array.isArray(v)) return {};
    return Object.fromEntries(Object.entries(v as Record<string, unknown>).filter((e): e is [string, string] => typeof e[1] === "string"));
  } catch {
    return {};
  }
}

export function readPlayPlan(date: string): string | null {
  return readAll()[date] ?? null;
}

export function writePlayPlan(date: string, choice: string | null): void {
  try {
    const all = readAll();
    if (choice) all[date] = choice.slice(0, 40);
    else delete all[date];
    const kept = Object.keys(all).sort().slice(-KEEP);
    (globalThis as { window?: { localStorage?: Storage } }).window?.localStorage?.setItem(PLAY_PLAN_KEY, JSON.stringify(Object.fromEntries(kept.map((k) => [k, all[k]]))));
  } catch {
    /* not remembered: fine */
  }
}

/** Moves one item (Góc học tập drag ⋮⋮, R6c). Invalid indexes: an unchanged copy. */
export function reorder<T>(list: readonly T[], from: number, to: number): T[] {
  const out = [...list];
  if (from < 0 || to < 0 || from >= out.length || to >= out.length || from === to) return out;
  const [x] = out.splice(from, 1);
  out.splice(to, 0, x as T);
  return out;
}
